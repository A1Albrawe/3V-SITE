import requests

def fetch_device_location_analytics():
    try:
        response = requests.get('https://ipapi.co').json()
        print(f"📡 [SUCCESS] City: {response.get('city')}")
        print(f"🌍 [SUCCESS] Country: {response.get('country_name')}")
    except Exception as e:
        print(f"❌ [ERROR] Radar Connect Failed: {e}")

if __name__ == '__main__':
    fetch_device_location_analytics()
